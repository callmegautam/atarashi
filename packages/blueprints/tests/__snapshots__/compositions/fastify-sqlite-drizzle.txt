# next steps
  Start the dev server:  pnpm run dev
  Copy `.env.example` to `.env` and fill in the blanks
  The database file is created on first run under `data/`
  Create the tables:  pnpm run db:push
  Browse the data:    pnpm run db:studio
  Check the API:  curl http://localhost:3000/health
  Lint the project:  pnpm run lint
  Run the tests:  pnpm run test
  Tests read the same environment as the app, so export your .env before running them

import { config } from 'dotenv';
import { z } from 'zod';

// Loads `.env` when one is present, before anything reads `process.env`. A
// missing file is not an error: in production the platform supplies the
// environment, and `dotenv` leaves whatever is already set untouched.
config({ quiet: true });

/**
 * The single source of truth for configuration. Every blueprint that needs an
 * environment variable contributes its line to the region below, so the shape
 * of `env` always matches `.env.example`.
 */
const schema = z.object({
  NODE_ENV: z.enum(['development', 'test', 'production']).default('development'),
  // #region atarashi:env-schema
  DATABASE_URL: z.string().min(1),
  PORT: z.coerce.number().int().positive().default(3000),
  // #endregion
});

const parsed = schema.safeParse(process.env);

if (!parsed.success) {
  console.error('Invalid environment variables:');
  for (const issue of parsed.error.issues) {
    console.error(`  - ${issue.path.join('.') || '(root)'}: ${issue.message}`);
  }
  process.exit(1);
}

export const env = parsed.data;
export type Env = typeof env;

import { mkdirSync } from 'node:fs';
import { dirname } from 'node:path';

import Database from 'better-sqlite3';

import { env } from '../config/env.js';

/** `file:./data/app.db` and `./data/app.db` both mean the same local file. */
const file = env.DATABASE_URL.replace(/^file:/, '');

mkdirSync(dirname(file), { recursive: true });

export const database = new Database(file);

database.pragma('journal_mode = WAL');
database.pragma('foreign_keys = ON');

/** Fails fast at boot rather than on the first request. */
export function checkDatabaseConnection() {
  database.prepare('SELECT 1').get();
}

import { integer, sqliteTable, text } from 'drizzle-orm/sqlite-core';

/** A starter table. Delete it once you have a real one. */
export const users = sqliteTable('users', {
  id: integer('id').primaryKey({ autoIncrement: true }),
  email: text('email').notNull().unique(),
  createdAt: integer('created_at', { mode: 'timestamp' })
    .notNull()
    .$defaultFn(() => new Date()),
  updatedAt: integer('updated_at', { mode: 'timestamp' }),
});

export type User = typeof users.$inferSelect;
export type NewUser = typeof users.$inferInsert;
