import { config } from 'dotenv';
import { z } from 'zod';

// Loads `.env` when one is present, before anything reads `process.env`. A
// missing file is not an error: in production the platform supplies the
// environment, and `dotenv` leaves whatever is already set untouched.
config({ quiet: true });

/**
 * The single source of truth for configuration. Every blueprint that needs an
 * environment variable contributes its line to the region below, so the shape
 * of `env` always matches `.env.example`.
 */
const schema = z.object({
  NODE_ENV: z.enum(['development', 'test', 'production']).default('development'),
  // #region atarashi:env-schema
  DATABASE_URL: z.string().min(1),
  PORT: z.coerce.number().int().positive().default(3000),
  // #endregion
});

const parsed = schema.safeParse(process.env);

if (!parsed.success) {
  console.error('Invalid environment variables:');
  for (const issue of parsed.error.issues) {
    console.error(`  - ${issue.path.join('.') || '(root)'}: ${issue.message}`);
  }
  process.exit(1);
}

export const env = parsed.data;
export type Env = typeof env;
